\chapter{Research Questions and Contributions}

\section{Thesis Title}
\label{sec:plan-title}

\textbf{Graph Neural Networks for Decentralized Multi-Agent Reinforcement
Learning in Power-Grid Topology Control}

\section{Central Thesis Question}
\label{sec:plan-question}

\latestrevision{\textbf{Which graph architecture and action-selection design
provide the most effective representation and control performance for
decentralized policies, and to what extent can a policy trained on a small grid
retain useful performance on a larger grid?}}

\section{Research Questions}
\label{sec:plan-research-questions}

\begin{enumerate}
  \item \textbf{Baseline learning:} Which training changes are needed for
  stable decentralized MAPPO topology control on \texttt{bus14}?
  Chapter~\ref{ch:baseline} establishes the reference through the training,
  initialization, and initial-action-bias experiments.

  \item \textbf{Sparse intervention:} Can the agents reduce unnecessary
  topology interventions without sacrificing source-grid survival?
  Chapter~\ref{ch:sparse-control} compares evaluation-time heuristics, a
  learned intervention gate, action penalties, adaptive budgets, and behavioral
  cloning on both survival and intervention frequency.

  \item \textbf{Graph representation:} Which graph representation,
  message-passing, graph-readout, and candidate-action design choices produce a
  useful local actor? Chapter~\ref{ch:methods} defines these choices and
  Chapter~\ref{ch:experiments} evaluates them on \texttt{bus14}.

  \newpage
  \item \textbf{Cross-grid transfer and adaptation:} To what extent do
  graph-actor weights learned on \texttt{bus14} retain useful performance on
  WCCI, and when zero-shot transfer is insufficient, does source initialization
  improve target-grid learning over training from scratch?
  {\color{red}Chapters~\ref{ch:encoder-inputs}--\ref{ch:screen-f-transfer}
  diagnose input compatibility, define the target protocol, test whether a
  frozen source encoder remains useful after replacing its action heads, and
  finally compare complete-policy zero-shot transfer with scratch training and
  fine-tuning.}
\end{enumerate}

\section{Contributions}
\label{sec:plan-contributions}

\begin{itemize}
  \item \textbf{A stabilized decentralized MAPPO reference on
  \texttt{bus14}.} Chapter~\ref{ch:baseline} isolates the training and
  initialization changes used by the later experiments.

  \item \textbf{A source-grid study of sparse intervention.}
  Chapter~\ref{ch:sparse-control} evaluates several mechanisms under the joint
  criteria of survival and intervention frequency rather than survival alone.

  \item \textbf{A shared graph actor with size-independent candidate-action
  scoring.} Chapters~\ref{ch:methods} and~\ref{ch:experiments} define the local
  graph, score actions from the nodes they affect, and screen representation,
  message-passing, graph-readout, and action-scoring choices.

  \item \textbf{A diagnosis and correction of cross-grid input mismatch.}
  Chapter~\ref{ch:encoder-inputs} measures what the shared encoder receives on
  both grids and motivates physically compatible power and angle inputs.

  \item \textbf{A controlled WCCI transfer and adaptation study.}
  {\color{red}Chapters~\ref{ch:wcci-transfer}--\ref{ch:screen-f-transfer}
  separate encoder-only transfer from complete-policy transfer, compare
  original and corrected input preprocessing, and evaluate zero-shot
  deployment, WCCI training from scratch, and source-initialized adaptation.}
\end{itemize}
